// vars/sharedPipelineGitNode.groovy
def call(Map config) {

    stage('Prepare Environment') {
        script {
            env.REPO_URL = config.GIT_URL
            env.BRANCH = config.BRANCH ?: 'main' // BRANCH가 없을 경우 'main'으로 설정
            env.PRODUCT_NAME = config.PRODUCT_NAME
            env.PRODUCT_VERSION = config.PRODUCT_VERSION
            env.PROJECT_NAME = config.PROJECT_NAME
            env.PROJECT_UUID = config.PROJECT_UUID

            echo "REPO_URL: ${env.REPO_URL}"
            echo "BRANCH: ${env.BRANCH}"
            echo "PRODUCT_NAME: ${env.PRODUCT_NAME}"
            echo "PRODUCT_VERSION: ${env.PRODUCT_VERSION}"
            echo "PROJECT_NAME: ${env.PROJECT_NAME}"
            echo "PROJECT_UUID: ${env.PROJECT_UUID}"
        }
    }

    stage('Checkout Source Code') {
        // Git 소스 코드 체크아웃
        checkout([$class: 'GitSCM', 
                  branches: [[name: "*/${env.BRANCH}"]], // 설정된 BRANCH를 사용
                  userRemoteConfigs: [[url: "${env.REPO_URL}", credentialsId: 'azure-devops-pat']]
        ])

        // 디버깅: 체크아웃 후 파일 확인
        sh 'ls -la $(pwd)'
        sh 'ls -la'
    }

    stage('Install Node.js and npm') {
        lock(resource: 'node-install', inversePrecedence: true, time: 10, unit: 'MINUTES') {
            sh '''
            sudo curl -fsSL https://deb.nodesource.com/setup_18.x | sudo bash -
            sudo apt-get install -y nodejs
            node --version
            npm --version
            '''
        }
    }

    if (fileExists('.npmrc')) {
        stage('Configure npm and Install Dependencies') {
            // npm 설치 잠금 설정
            lock(resource: 'npm-install', inversePrecedence: true, time: 10, unit: 'MINUTES') {
                script {
                    retry_count = 0
                    max_retries = 5
                    success = false

                    while (retry_count < max_retries) {
                        try {
                            echo "Attempt ${retry_count + 1} to install npm dependencies..."

                            sh '''
                            # AZURE_PAT을 base64로 인코딩
                            echo "Using AZURE_PAT (first 4 chars): $(echo $AZURE_PAT | cut -c1-4)****"
                            ENCODED_PAT=$(echo -n $AZURE_PAT | base64 | tr -d '\n')
                            echo "Encoded PAT (first 4 chars): $(echo $ENCODED_PAT | cut -c1-4)****"

                            # .npmrc 파일 설정
                            cp .npmrc ~/.npmrc
                            sed -i "s|_password=.*|_password=${ENCODED_PAT}|" ~/.npmrc
                            cat ~/.npmrc  # 디버깅용

                            # npm 의존성 설치
                            npm install --loglevel verbose --prefer-offline --cache ~/.npm --package-lock --legacy-peer-deps || echo "Ignoring npm install failure"
                            '''
                            success = true
                            break
                        } catch (Exception err) {
                            echo "Failed to install npm dependencies: ${err}. Retrying in 30 seconds..."
                            retry_count += 1
                            sleep 30
                        }
                    }

                    if (!success) {
                        error "Exceeded maximum number of retries (${max_retries}). Failing the task."
                    }
                }
            }
        }
    }

    stage('Install Trivy') {
        lock(resource: 'apt-get', inversePrecedence: true, time: 10, unit: 'MINUTES') {
            script {
                // 설치된 Trivy 버전 확인
                def installedVersion = sh(script: "trivy --version | grep 'Version:' | awk '{print \$2}' || echo 'none'", returnStdout: true).trim()
                echo "Installed Trivy version: ${installedVersion}"

                def latestVersion = sh(script: '''
                    wget -qO- https://api.github.com/repos/aquasecurity/trivy/releases/latest | jq -r .tag_name | sed 's/^v//'
                ''', returnStdout: true).trim()
                echo "Latest Trivy version: ${latestVersion}"

                if (installedVersion == 'none' || installedVersion != latestVersion) {
                    echo "Installing Trivy version ${latestVersion}..."
                    sh """
                    sudo apt-get update
                    sudo apt-get install -y wget jq
                    TRIVY_DOWNLOAD_URL="https://github.com/aquasecurity/trivy/releases/download/v${latestVersion}/trivy_${latestVersion}_Linux-64bit.deb"
                    wget \${TRIVY_DOWNLOAD_URL}
                    sudo dpkg -i trivy_*.deb
                    trivy --version
                    """
                } else {
                    echo "Trivy is up-to-date."
                }
            }
        }
    }


    stage('Generate SBOM with Trivy') {
        // Trivy로 SBOM 생성
        sh 'trivy fs --format cyclonedx --include-dev-deps --output sbom.json .'
    }

    stage('Display SBOM Size') {
        script {
            // SBOM 파일 크기 확인
            def fileSize = sh(script: "stat --printf='%s' sbom.json", returnStdout: true).trim()
            echo "SBOM file size: ${fileSize} bytes"
        }
    }

    stage('Check Azure CLI Version') {
        script {
            try {
                sh 'az --version'
                echo 'Azure CLI is already installed.'
            } catch (Exception e) {
                error "Azure CLI is not installed or could not be found. Please ensure it's included in the Docker image."
            }
        }
    }

    stage('Upload SBOM to Azure Blob Storage') {
        script {
            // Azure Blob Storage로 SBOM 업로드
            def storagePath = "${env.PRODUCT_NAME}/${env.PRODUCT_VERSION}/${new Date().format('yyyyMMdd')}/${env.PRODUCT_NAME}_${env.PRODUCT_VERSION}_${env.PROJECT_NAME}.sbom.json"
            
            sh """
            az storage blob upload --account-name ${env.ACCOUNT_NAME} --account-key ${env.ACCOUNT_KEY} \
            --container-name ${env.CONTAINER_NAME} --name ${storagePath} --file sbom.json --overwrite
            """
            echo "SBOM uploaded to Azure Blob Storage at: ${storagePath}"
        }
    }

    stage('Upload SBOM to Dependency Track') {
        script {
            // Dependency Track로 SBOM 업로드
            sh '''
            curl -X POST \
            -H "X-Api-Key: $DTRACK_API_KEY" \
            -F "project=$PROJECT_UUID" \
            -F "bom=@sbom.json" \
            $DTRACK_API_URL
            '''
        }
    }

    stage('Archive SBOM') {
        script {
            // 파일 이름 변경 및 아티팩트 보관
            def sbomFileName = "${env.PRODUCT_NAME}_${env.PRODUCT_VERSION}_${env.PROJECT_NAME}.sbom.json"
            sh "mv sbom.json ${sbomFileName}"
            archiveArtifacts artifacts: sbomFileName, allowEmptyArchive: true
        }
    }
}
