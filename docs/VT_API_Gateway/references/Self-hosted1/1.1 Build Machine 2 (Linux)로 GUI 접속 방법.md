# TigerVNC 이용

- VNC서버
    - TigerVNC 서버 설치
        - sudo apt updatesudo apt install -y tigervnc-standalone-server tigervnc-common xfce4 xfce4-goodies dbus-x11
    - VNC 비밀번호 설정
        - vncpasswd
        - 비밀번호로 Ewoosoft1!@#입력
        - Would you like to enter a view-only password (y/n)? n
    - VNC 서버 시작 스크립트 생성
        - mkdir -p ~/.vnc
            
            cat > ~/.vnc/xstartup << EOF
            
            #!/bin/sh
            
            unset SESSION_MANAGER
            
            unset DBUS_SESSION_BUS_ADDRESS
            
            export XDG_SESSION_TYPE=x11
            
            export DISPLAY=:4
            
            export GNOME_SHELL_SESSION_MODE=ubuntu
            
            export XDG_CURRENT_DESKTOP=ubuntu:GNOME
            
            # 화면 잠금 비활성화
            
            gsettings set org.gnome.desktop.lockdown disable-lock-screen true
            
            gsettings set org.gnome.desktop.screensaver lock-enabled false
            
            # 자동 로그인 설정
            
            dbus-launch --exit-with-session gnome-session
            
            EOF
            
            chmod +x ~/.vnc/xstartup
            
    - Xauthority 파일 생성
        - touch ~/.Xauthoritychmod 600 ~/.Xauthority
    - VNC 서버 시작
        - vncserver :4 -localhost no -depth 24 -geometry 1920x1080
    - 서버 시작 확인
        - vncserver -list
            
            TigerVNC server sessions:

            4               5902                            135326          Xtigervnc
            
    - 방화벽 설정
        - sudo ufw allow 5901sudo ufw status
    - Linux Rebooting해도 자동 실행하게 설정하기
        - systemd 서비스 파일 생성
            - sudo vi /etc/systemd/system/vncserver-admin.service
                
                [Unit]
                
                Description=VNC Server for admin
                
                After=syslog.target network.target
                
                [Service]
                
                Type=forking
                
                User=admin
                
                Group=admin
                
                WorkingDirectory=/home/admin
                
                Environment=HOME=/home/admin
                
                Environment=USER=admin
                
                Environment=DISPLAY=:4
                
                Environment=XDG_SESSION_TYPE=x11
                
                Environment=GNOME_SHELL_SESSION_MODE=ubuntu
                
                Environment=XDG_RUNTIME_DIR=/run/user/1001
                
                ExecStartPre=/bin/sh -c '/usr/bin/vncserver -kill :4 > /dev/null 2>&1 || :'
                
                ExecStart=/usr/bin/vncserver :4 -geometry 1920x1080 -depth 24 -localhost no
                
                ExecStop=/usr/bin/vncserver -kill :4
                
                Restart=on-failure
                
                RestartSec=3
                
                [Install]
                
                WantedBy=multi-user.target
                
        - 서비스 활성화
            - sudo systemctl daemon-reload
                
                sudo systemctl enable vncserver-admin.service
                
                sudo systemctl start vncserver-admin.service
